# Записывать так не удобно, список имен кошек это же СПИСОК, зачем пложить переменные, давай сделаем цикл input который будет заполнять именами единый список catNames




catNames = []  				# Объявляем переменную с пустым списком
while True:
    print('Укажите имя кота или кошки', + str(len(catNames) + 1) + '(<ENTER> для завершения:') # выводим сообщение для пользователя
    name = input() 			# объявляем переменную для ввода имени
    if name = '':		# Если name пусто, то заканчиваем бесконечный цикл while
        break
    catNames = catNames + name 	#Конкатенаия списка токесть добавляем в спиок имя которое ввел юзер 
print('Имена котиков и кошек: ')
for name in catNames:    	# Запускаем цикл перечисленя значений name из списка catNames
    print '  ' + name)
